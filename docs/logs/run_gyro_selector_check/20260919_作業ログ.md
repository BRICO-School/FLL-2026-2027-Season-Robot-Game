+"""
+【セレクターと同じ形で、ミッションごとのジャイロの状態を確かめる】（2026-09-19）
+
+競技では selector.py がハブの中で動きつづけ、initialize_robot()（ジャイロの待ち 2 秒をふくむ）は最初の 1 回だけ。
+ミッションごとには reset_robot()（距離と向きを 0 に戻す）しかしない。
+このとき、ミッションのあいだに機体を持ち上げたり運んだりしても、ジャイロは状態 B（正確な目盛り）のままか、
+始めてすぐの状態 A（1 周を約 365° と数える）に戻ってしまうかを確かめる。
+
+やり方（1 回のプログラムの中で、何回でもくり返せる）:
+ 1. 実行すると、ライトが白で待つ（セレクターのボタン待ちと同じ）
+ 2. 試したい扱い方をする（例: そのまま置いておく／持ち上げて振ってから置く／ホームへ運ぶまねをする）
+ 3. 機体の左側面を定規に当てて置き、手を離して **ハブの右ボタン（▶）** を押す → すぐ 5 周まわる
+ 4. 緑（ピー）になったら定規に当て直して手を離す → 青 → 1 周の読みが出る → 白に戻る（2 へ）
+ 5. やめるときはハブの中央ボタン
+
+出る行:  G, 回, 押すまで待った時間 ms, 待つあいだ静止と判定された割合 %, 当て直しで動いた角度, 1 周の読み, 校正前の目盛りで
+ 校正前の目盛りで 約 360 なら状態 B・約 365 なら状態 A。
+
+SETTLE = False にすると、initialize_robot() のジャイロの待ち（2 秒）を飛ばして始める
+（待ちを無くしても、ボタン待ちの時間だけで足りるかを見るため）。
+ハブの設定は書きかえない。
+"""
+
+import setup
+from pybricks.parameters import Button, Color
+from pybricks.tools import StopWatch, run_task, wait
+from setup import initialize_robot
+
+TURNS = 5
+SETTLE = True  # False: 始めのジャイロの待ち（setup.GYRO_SETTLE_MS）を 0 にして始める
+
+
+async def realign(hub):
+    """緑にして、手で定規に当て直されるのを待ち、当て直したあとのジャイロの向きを返す"""
+    hub.light.on(Color.GREEN)
+    hub.speaker.volume(100)
+    await hub.speaker.beep(frequency=500, duration=600)
+    total = StopWatch()
+    still = StopWatch()
+    await wait(1500)  # 手を伸ばす時間
+    still.reset()
+    while True:
+        if not hub.imu.stationary():
+            still.reset()
+        if total.time() > 6000 and still.time() > 2500:  # 少なくとも 6 秒は待つ
+            break
+        await wait(20)
+    h2 = hub.imu.heading()
+    hub.light.on(Color.BLUE)
+    await hub.speaker.beep(frequency=1000, duration=200)
+    return h2
+
+
+async def run(hub, robot, left_wheel, right_wheel, left_lift, right_lift):
+    print("# imu.settings:", hub.imu.settings(), "/ 始めのジャイロの待ち:", SETTLE)
+    count = 0
+    while True:
+        # ----- セレクターのボタン待ちと同じ: 押されるまで待つ -----
+        hub.light.on(Color.WHITE)
+        waited = StopWatch()
+        samples = 0
+        still_samples = 0
+        while Button.RIGHT not in hub.buttons.pressed():
+            samples += 1
+            if hub.imu.stationary():
+                still_samples += 1
+            await wait(20)
+        waited_ms = waited.time()
+
+        # ----- selector.py の reset_robot() と同じ準備をして、すぐ走り出す -----
+        hub.light.on(Color.RED)
+        robot.stop()
+        robot.reset()
+        hub.imu.reset_heading(0)
+        await wait(50)
+        h0 = hub.imu.heading()
+        await robot.turn(360 * TURNS)
+        await wait(1000)
+        h1 = hub.imu.heading()
+        robot.stop()  # モーターの力を抜く（手で回せるように）
+        h2 = await realign(hub)
+
+        count += 1
+        per_turn = abs(h2 - h0) / TURNS
+        print(
+            "G,",
+            count,
+            ", 押すまで",
+            waited_ms,
+            "ms , 静止",
+            round(100.0 * still_samples / max(samples, 1), 0),
+            "% , 当て直しで動いた角度 h2-h1",
+            round(h2 - h1, 2),
+            ", 1 周の読み",
+            round(per_turn, 3),
+            ", 校正前の目盛りで",
+            round(per_turn * hub.imu.settings()[-1] / 360, 2),
+            ", 電池",
+            hub.battery.voltage(),
+        )
+
+
+def main(settle=None):
+    global SETTLE
+    if settle is not None:
+        SETTLE = settle
+    if not SETTLE:
+        setup.GYRO_SETTLE_MS = 0  # 待ちを飛ばす（この道具の中だけ）
+    hub, robot, left_wheel, right_wheel, left_lift, right_lift = initialize_robot()
+    run_task(run(hub, robot, left_wheel, right_wheel, left_lift, right_lift))
+
+
+if __name__ == "__main__":
+    main()
